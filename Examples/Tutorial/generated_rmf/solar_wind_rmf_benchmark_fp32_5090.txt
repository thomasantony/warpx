# Explicit open-boundary solar-wind RMF run with grid-driven coil currents.
#
# Two smooth circular current loops are centered at the origin. One loop is in
# the x-z plane and drives By ~ cos(omega t); the other is in the x-y plane and
# drives Bz ~ sin(omega t). The imposed source is a grid current, so Maxwell's
# equations evolve the launched fields and the plasma current self-consistently.

my_constants.v_sw = 4.0e5
my_constants.beta_sw = v_sw / clight
my_constants.n_sw = 5.0e6
my_constants.flux_sw = n_sw * v_sw
my_constants.inflow_ppc = 0.05

my_constants.B0 = 1.0e-5
my_constants.R_coil = 20.0
my_constants.w_coil = 8.0
my_constants.r_eps = 1.0e-12
my_constants.f_rmf = 7.0e2
my_constants.omega_rmf = 2.0*pi*f_rmf
my_constants.I_coil = 2.0*R_coil*B0/mu0
my_constants.J0_coil = I_coil/(2.0*pi*w_coil*w_coil)
my_constants.t_ramp = 1.0e-7
my_constants.dt = 1.0e-9

#################################
####### GENERAL PARAMETERS ######
#################################
max_step = 2000

geometry.dims = 3
geometry.prob_lo = -100.0 -100.0 -100.0
geometry.prob_hi =  100.0  100.0  100.0
amr.n_cell = 128 128 128
amr.max_grid_size = 16
amr.blocking_factor = 8
amr.max_level = 0

#################################
####### Boundary condition ######
#################################
boundary.field_lo = pml pml pml
boundary.field_hi = pml pml pml
warpx.pml_ncell = 8

boundary.particle_lo = absorbing absorbing absorbing
boundary.particle_hi = absorbing absorbing absorbing

#################################
############ NUMERICS ###########
#################################
warpx.serialize_initial_conditions = 1
warpx.verbose = 1
warpx.const_dt = dt
warpx.use_filter = 1

algo.maxwell_solver = Yee
algo.particle_shape = 2
algo.current_deposition = villasenor
algo.field_gathering = energy-conserving

warpx.do_dive_cleaning = 0
warpx.do_divb_cleaning = 0

#################################
########### RMF COILS ###########
#################################
# Smooth toroidal current tubes approximate finite coils on the grid. A short
# raised-cosine current ramp avoids launching an artificial turn-on transient.
# The y-normal coil lies in the x-z plane. The z-normal coil lies in the x-y
# plane. The two drives are 90 degrees out of phase.
warpx.Jx_external_grid_function(x,y,z,t) = "if(t<t_ramp,0.5*(1.0-cos(pi*t/t_ramp)),1.0)*(-J0_coil*z/sqrt(x*x+z*z+r_eps)*exp(-((sqrt(x*x+z*z)-R_coil)**2+y*y)/(2.0*w_coil*w_coil))*cos(omega_rmf*t) - J0_coil*y/sqrt(x*x+y*y+r_eps)*exp(-((sqrt(x*x+y*y)-R_coil)**2+z*z)/(2.0*w_coil*w_coil))*sin(omega_rmf*t))"
warpx.Jy_external_grid_function(x,y,z,t) = "if(t<t_ramp,0.5*(1.0-cos(pi*t/t_ramp)),1.0)*J0_coil*x/sqrt(x*x+y*y+r_eps)*exp(-((sqrt(x*x+y*y)-R_coil)**2+z*z)/(2.0*w_coil*w_coil))*sin(omega_rmf*t)"
warpx.Jz_external_grid_function(x,y,z,t) = "if(t<t_ramp,0.5*(1.0-cos(pi*t/t_ramp)),1.0)*J0_coil*x/sqrt(x*x+z*z+r_eps)*exp(-((sqrt(x*x+z*z)-R_coil)**2+y*y)/(2.0*w_coil*w_coil))*cos(omega_rmf*t)"

#################################
############ PLASMA #############
#################################
particles.species_names = electrons protons

electrons.charge = -q_e
electrons.mass = m_e
electrons.injection_sources = initial_electrons inflow_electrons

# Initial neutral solar-wind fill, so the 200 m box does not need to fill from
# the inflow boundary over a 0.5 ms transit time.
electrons.initial_electrons.injection_style = NUniformPerCell
electrons.initial_electrons.num_particles_per_cell_each_dim = 2 2 2
electrons.initial_electrons.profile = constant
electrons.initial_electrons.density = n_sw
electrons.initial_electrons.momentum_distribution_type = gaussian
electrons.initial_electrons.ux_m = beta_sw
electrons.initial_electrons.ux_th = 4.4e-3
electrons.initial_electrons.uy_th = 4.4e-3
electrons.initial_electrons.uz_th = 4.4e-3

# Continuous upstream solar-wind flux through the -x plane.
electrons.inflow_electrons.injection_style = NFluxPerCell
electrons.inflow_electrons.surface_flux_pos = -100.0
electrons.inflow_electrons.flux_normal_axis = x
electrons.inflow_electrons.flux_direction = 1
electrons.inflow_electrons.num_particles_per_cell = inflow_ppc
electrons.inflow_electrons.flux_profile = constant
electrons.inflow_electrons.flux = flux_sw
electrons.inflow_electrons.momentum_distribution_type = gaussianflux
electrons.inflow_electrons.ux_m = beta_sw
electrons.inflow_electrons.ux_th = 4.4e-3
electrons.inflow_electrons.uy_th = 4.4e-3
electrons.inflow_electrons.uz_th = 4.4e-3

protons.charge = q_e

protons.inflow_protons.surface_flux_pos = -100.0
protons.inflow_protons.flux_normal_axis = x
protons.inflow_protons.flux_direction = 1
protons.inflow_protons.num_particles_per_cell = inflow_ppc
protons.inflow_protons.flux_profile = constant
protons.inflow_protons.flux = flux_sw
protons.inflow_protons.momentum_distribution_type = gaussianflux
protons.inflow_protons.ux_m = beta_sw
protons.inflow_protons.ux_th = 1.5e-4
protons.inflow_protons.uy_th = 1.5e-4
protons.inflow_protons.uz_th = 1.5e-4
